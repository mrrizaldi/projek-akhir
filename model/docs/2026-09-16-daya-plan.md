# HW-7 Pengukuran Daya — Rencana Implementasi

> **Untuk pekerja agentik:** SUB-SKILL WAJIB: pakai
> `superpowers:subagent-driven-development` (disarankan) atau
> `superpowers:executing-plans` untuk mengerjakan rencana ini task demi task.
> Langkah memakai checkbox (`- [ ]`) untuk penanda kemajuan.

**Goal:** Mengukur konsumsi daya ESP32-S3 per tahap kerja (akuisisi / DSP /
inferensi / LED) dengan INA219, lalu menurunkannya jadi rekomendasi kapasitas
baterai.

**Arsitektur:** Firmware mendapat tangga mode `p0`–`p4` yang dipilih lewat
serial; tiap mode mematikan satu lapis pekerjaan, jadi ongkos tiap lapis =
selisih antar-mode. Sinyal uji di-replay dari `golden_raw` supaya bebannya
identik tiap ulangan. INA219 dibaca DUT sendiri lewat I2C tanpa library pihak
ketiga — register kalibrasi on-chip dilewati, konversi dilakukan di software
sebagai fungsi murni yang bisa diuji di PC.

**Tech Stack:** C++ (PlatformIO, Arduino framework, ESP32-S3), Unity (test
native), Python 3.11 + pyserial + numpy + matplotlib, pytest, Make.

**Spek:** `model/docs/2026-09-16-daya-design.md`

```cpp
// Register kalibrasi on-chip SENGAJA tidak dipakai (lihat

// supaya R_SHUNT jadi knob kalibrasi yang kelihatan.
#include <unity.h>
#include "ina219.h"

static void test_arus_positif(void)
{
    // 1000 LSB x 0,01 mV = 10 mV di shunt; 10 mV / 0,1 ohm = 100 mA
    TEST_ASSERT_FLOAT_WITHIN(0.01f, 100.0f, ina219_arus_mA(1000, 0.1f));
}

static void test_arus_negatif_saat_terbalik(void)
{
    // VIN+ dan VIN- tertukar: arus terbaca negatif, bukan rusak.
    TEST_ASSERT_FLOAT_WITHIN(0.01f, -100.0f, ina219_arus_mA(-1000, 0.1f));
}

static void test_arus_nol(void)
{
    TEST_ASSERT_FLOAT_WITHIN(0.001f, 0.0f, ina219_arus_mA(0, 0.1f));
}

static void test_shunt_lain_menggeser_skala(void)
{
    // R_SHUNT adalah knob kalibrasi: shunt 2x lebih besar -> arus 2x lebih kecil
    TEST_ASSERT_FLOAT_WITHIN(0.01f, 50.0f, ina219_arus_mA(1000, 0.2f));
}

{
    // 1250 x 4 mV = 5,0 V; nilainya ada di bit 15:3

{

}

void setUp(void) {}
```

```cpp
// Konversi register INA219 -> satuan teknik. Murni, tanpa I2C dan tanpa
// Arduino, supaya bisa diuji di PC (pio test -e native).
//
// Register kalibrasi on-chip tidak dipakai: R_SHUNT dibagi di software supaya
// jadi knob kalibrasi eksplisit terhadap multimeter.
#ifndef INA219_H
#define INA219_H

#include <stdint.h>

#ifdef __cplusplus
extern "C" {
#endif

#define INA219_ALAMAT     0x40      // A0=A1=GND (bawaan modul)
#define INA219_REG_CONFIG 0x00
#define INA219_REG_SHUNT  0x01
#define INA219_REG_BUS    0x02

// BRNG=16V, PGA=/1 (+-40 mV), BADC & SADC = rata-rata 128 sampel (~68 ms),
// MODE = shunt+bus kontinu. Averaging WAJIB: arus alat ini berdenyut mengikuti
// burst 33 ms (HW-3), pembacaan sesaat melompat tanpa arti.
#define INA219_CONFIG     0x07FF

// Shunt voltage: 1 LSB = 0,01 mV, signed. mV / ohm = mA.
float ina219_arus_mA(int16_t shunt_reg, float r_shunt_ohm);

// Bus voltage: nilai di bit 15:3, 1 LSB = 4 mV. Bit 1 (CNVR) & 0 (OVF) dibuang.
float ina219_bus_V(uint16_t bus_reg);

#ifdef __cplusplus
}
#endif
#endif  // INA219_H
```

Buat `firmware/src/ina219.cpp`:

```cpp
#include "ina219.h"

float ina219_arus_mA(int16_t shunt_reg, float r_shunt_ohm)
{
    return ((float)shunt_reg * 0.01f) / r_shunt_ohm;
}

float ina219_bus_V(uint16_t bus_reg)
{
    return (float)(bus_reg >> 3) * 0.004f;
}
```

```cpp
static void ina_tulis(uint8_t reg, uint16_t nilai)
{
    Wire.beginTransmission(INA219_ALAMAT);
    Wire.write(reg);
    Wire.write((uint8_t)(nilai >> 8));
    Wire.write((uint8_t)(nilai & 0xFF));
    Wire.endTransmission();
}

static uint16_t ina_baca(uint8_t reg)
{
    Wire.beginTransmission(INA219_ALAMAT);
    Wire.write(reg);
    Wire.endTransmission(false);            // repeated start, bukan stop
    if (Wire.requestFrom((int)INA219_ALAMAT, 2) != 2) return 0;
    const uint8_t hi = Wire.read(), lo = Wire.read();
    return ((uint16_t)hi << 8) | lo;
}

// Return true kalau modul menjawab. Config ditulis lalu dibaca balik: modul
// yang tidak ada membalas 0x0000 atau 0xFFFF, dua-duanya != INA219_CONFIG.
static bool ina_siapkan()
{
    Wire.begin(PIN_SDA, PIN_SCL);
    ina_tulis(INA219_REG_CONFIG, INA219_CONFIG);
    delay(5);
    return ina_baca(INA219_REG_CONFIG) == INA219_CONFIG;
}

static void ina_lapor()
{
    if (!ina_ada) { Serial.println("INA219 tidak terdeteksi di 0x40."); return; }
    const float i = ina219_arus_mA((int16_t)ina_baca(INA219_REG_SHUNT), R_SHUNT);
    const float v = ina219_bus_V(ina_baca(INA219_REG_BUS));
    Serial.printf("daya | I=%.1f mA | V=%.3f V | P=%.1f mW\n", i, v, i * v);
}
```

```python
"""HW-7 — ukur konsumsi daya per tahap, lalu turunkan kapasitas baterai.

    make daya        (atau: .venv/bin/python scripts/measure_power.py)

Board dijalankan di lima mode berurutan (p0..p4), tiap mode 60 detik, dan
arusnya dicatat dari INA219 lewat serial. Ongkos tiap tahap = selisih
antar-mode. Rancangan lengkap: docs/2026-09-16-daya-design.md

Port lain: ECG_PORT=/dev/ttyACM1 make daya
Durasi lain: DAYA_DETIK=30 make daya
"""
import datetime
import os
import re
import sys
import time

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt          # noqa: E402
import numpy as np                        # noqa: E402

PORT = os.environ.get("ECG_PORT", "/dev/ttyACM0")
BAUD = 115200
DETIK = int(os.environ.get("DAYA_DETIK", "60"))
MODE = ["p0", "p1", "p2", "p3", "p4"]
LABEL = {
    "p0": "board idle",
    "p1": "+ akuisisi ISR",
    "p2": "+ DSP (bandpass/deteksi)",
    "p3": "+ inferensi TFLM",
    "p4": "+ LED",
}
ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
TUJUAN = os.path.join(ROOT, "artifacts", "metrics")

# Asumsi yang DITULIS sebagai asumsi, bukan fakta (docs §6).
V_BAT = 3.7          # LiPo nominal
ETA = 0.85           # efisiensi regulator dev board — tidak ada di datasheet
TERPAKAI = 0.8       # LiPo jangan dikuras habis

BARIS = re.compile(r"^DAYA (p\d) ([-\d.]+) ([\d.]+) ([-\d.]+) (\d+) (\d+)")


def rata_per_mode(baris):
    """[(mode, mA), ...] -> {mode: (rata, std)}."""
    keluar = {}
    for m in MODE:
        nilai = [a for mm, a in baris if mm == m]
        if nilai:
            keluar[m] = (float(np.mean(nilai)), float(np.std(nilai)))
    return keluar


def cek_tangga(rata, faktor_derau=2.0):
    """Dua assert yang benar-benar bisa gagal (docs §5).

    Sengaja BUKAN 'jumlah selisih == p3-p0': itu teleskopik, selalu benar,
    tidak pernah bisa gagal.
    """
    ada = [m for m in MODE if m in rata]
    for a, b in zip(ada, ada[1:]):
        delta = rata[b][0] - rata[a][0]
        assert delta >= 0, (
            f"arus turun dari {a} ({rata[a][0]:.1f} mA) ke {b} "
            f"({rata[b][0]:.1f} mA) — gating mode bocor")
        derau = faktor_derau * max(rata[a][1], rata[b][1])
        assert delta > derau, (
            f"selisih {a}->{b} cuma {delta:.2f} mA, tenggelam dalam derau "
            f"{derau:.2f} mA — tahap ini belum terukur")


def kapasitas_mAh(p_mw, jam, v_bat=V_BAT, eta=ETA, terpakai=TERPAKAI):
    """Daya [mW] -> kapasitas baterai [mAh].

    Lewat DAYA, bukan arus: mA di 5 V bukan mA di 3,7 V.
    """
    i_bat = p_mw / (v_bat * eta)
    return i_bat * jam / terpakai


def _kumpulkan(s):
    baris, mentah = [], []
    for m in MODE:
        s.write(m[1].encode())            # 'p3' -> kirim '3'
        s.flush()
        print(f"  {m} ({LABEL[m]}) — {DETIK} detik...", flush=True)
        habis = time.time() + DETIK
        n = 0
        while time.time() < habis:
            t = s.readline().decode(errors="replace").strip()
            c = BARIS.match(t)
            if not c:
                continue
            if c.group(1) != m:
                continue                  # baris sisa dari mode sebelumnya
            if n >= 3:                    # buang 3 detik pertama: transien
                baris.append((m, float(c.group(2))))
                mentah.append((m, float(c.group(2)), float(c.group(3)),
                               float(c.group(4)), int(c.group(5))))
            n += 1
        hilang = [r[4] for r in mentah if r[0] == m]
        if hilang and max(hilang) > 0:
            sys.exit(f"{m}: {max(hilang)} sampel hilang — beban tidak sahih, "
                     "ulangi pengukuran")
    return baris, mentah


def main() -> None:
    try:
        import serial
    except ImportError:
        sys.exit("pyserial tidak ada. Pasang: .venv/bin/pip install -r requirements.txt")

    try:
        s = serial.Serial(PORT, BAUD, timeout=2)
    except Exception as e:                                   # noqa: BLE001
        sys.exit(f"Tidak bisa membuka {PORT}: {e}\n"
                 "Serial monitor lain masih terbuka? Cek: lsof /dev/ttyACM0")

    time.sleep(1.0)
    s.reset_input_buffer()
    print(f"Mengukur {len(MODE)} mode x {DETIK} detik "
          f"(~{len(MODE) * DETIK // 60} menit)...")

    baris, mentah = _kumpulkan(s)
    s.write(b"n")                         # kembalikan board ke operasi normal
    s.flush()
    s.close()

    if not baris:
        sys.exit("Tidak ada baris DAYA. Firmware HW-7 sudah di-flash? "
                 "INA219 terdeteksi saat boot?")

    rata = rata_per_mode(baris)
    cek_tangga(rata)

    cap = datetime.datetime.now().strftime("%Y%m%d-%H%M")
    os.makedirs(TUJUAN, exist_ok=True)
    csv = os.path.join(TUJUAN, f"daya_{cap}.csv")
    with open(csv, "w") as f:
        f.write("mode,mA,V,mW,lewat\n")
        for r in mentah:
            f.write(f"{r[0]},{r[1]:.2f},{r[2]:.4f},{r[3]:.2f},{r[4]}\n")

    print(f"\n{'mode':4} {'yang jalan':28} {'mA':>8} {'std':>6} {'selisih':>9}")
    sebelum = None
    for m in MODE:
        if m not in rata:
            continue
        a, sd = rata[m]
        d = "" if sebelum is None else f"{a - sebelum:+8.2f}"
        print(f"{m:4} {LABEL[m]:28} {a:8.2f} {sd:6.2f} {d:>9}")
        sebelum = a

    p3_mw = rata["p3"][0] * np.mean([r[2] for r in mentah if r[0] == "p3"])
    print(f"\nDaya operasi (p3): {p3_mw:.1f} mW")
    print(f"Asumsi: V_bat {V_BAT} V, eta {ETA}, terpakai {TERPAKAI}")
    print(f"{'target jam':>12} {'kapasitas baterai':>20}")
    for jam in (8, 12, 24):
        print(f"{jam:>12} {kapasitas_mAh(p3_mw, jam):>17.0f} mAh")

    png = os.path.join(TUJUAN, f"daya_{cap}.png")
    fig, ax = plt.subplots(figsize=(8, 4.5))
    ada = [m for m in MODE if m in rata]
    ax.bar(ada, [rata[m][0] for m in ada],
           yerr=[rata[m][1] for m in ada], capsize=4, color="#4878a8")
    ax.set_ylabel("arus rata-rata (mA)")
    ax.set_title(f"HW-7 konsumsi per tahap — {cap}")
    for i, m in enumerate(ada):
        ax.text(i, rata[m][0], f"{rata[m][0]:.1f}", ha="center", va="bottom")
    fig.tight_layout()
    fig.savefig(png, dpi=120)
    print(f"\nCSV: {csv}\nPNG: {png}")


if __name__ == "__main__":
    main()
```
